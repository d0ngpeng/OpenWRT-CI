# ============================================================
# 京东云 AX6600（雅典娜 / RE-CS-02 / IPQ6010 + QCN9074 三 radio）
# 源码：laipeng668/immortalwrt @ openwrt-25.12（ImmortalWrt + 满血 NSS）
# ============================================================

#设备平台
CONFIG_TARGET_qualcommax=y
CONFIG_TARGET_qualcommax_ipq60xx=y

#设备列表：只编 AX6600，省时间也避免刷错包
CONFIG_TARGET_DEVICE_qualcommax_ipq60xx_DEVICE_jdcloud_re-cs-02=y

#设备额外包：雅典娜 LED 控制（包由 Scripts/Packages.sh 拉取）
#这一行只在 CONFIG_TARGET_PER_DEVICE_ROOTFS=y 时生效，作用是在设备自带的
#DEVICE_PACKAGES（ipq-wifi-jdcloud_re-cs-02 + ath11k-firmware-qcn9074）之外再追加，
#不会覆盖，放心加。
CONFIG_TARGET_DEVICE_PACKAGES_qualcommax_ipq60xx_DEVICE_jdcloud_re-cs-02="luci-app-athena-led luci-i18n-athena-led-zh-cn"
CONFIG_PACKAGE_luci-app-athena-led=m
CONFIG_PACKAGE_luci-i18n-athena-led-zh-cn=m

#内核：6.18（NSS 分支的 testing kernel，GWRT V2 用的同一条线）
CONFIG_TESTING_KERNEL=y

#AX6600 是 1G 内存 + 三 radio，ath11k 必须用 512M 的 Q6 预留，
#否则会出现只有 2.4G、5G 起不来或者无线直接不初始化
CONFIG_ATH11K_MEM_PROFILE_512M=y

# ============================================================
# NSS 硬件加速
# 光选 kmod 包不够：nss-packages 里每个 client 模块还挂在 NSS_DRV_*_ENABLE
# 这些开关上，不开就是「包在 .config 里、实际被 Kconfig 隐藏」，最后编出来
# 是个纯软转发的系统。下面这些必须显式打开。
# ============================================================

#NSS 固件 12.5（和 nss-packages 的 qca-nss-ecm/qca-nss-drv 对齐）
CONFIG_NSS_FIRMWARE_VERSION_12_5=y

#NSS 数据面 + 核心驱动 + ECM 连接管理
CONFIG_PACKAGE_kmod-qca-nss-dp=y
CONFIG_PACKAGE_kmod-qca-nss-drv=y
CONFIG_PACKAGE_kmod-qca-nss-ecm=y
CONFIG_PACKAGE_kmod-qca-nss-crypto=y

#NSS 功能开关（对应下面这些 client 模块）
CONFIG_NSS_DRV_BRIDGE_ENABLE=y
CONFIG_NSS_DRV_IPV6_ENABLE=y
CONFIG_NSS_DRV_VIRT_IF_ENABLE=y
CONFIG_NSS_DRV_VLAN_ENABLE=y
CONFIG_NSS_DRV_PPPOE_ENABLE=y
CONFIG_NSS_DRV_PPTP_ENABLE=y
CONFIG_NSS_DRV_L2TP_ENABLE=y
CONFIG_NSS_DRV_GRE_ENABLE=y
CONFIG_NSS_DRV_IGS_ENABLE=y
CONFIG_NSS_DRV_SHAPER_ENABLE=y
CONFIG_NSS_DRV_LAG_ENABLE=y
CONFIG_NSS_DRV_MAPT_ENABLE=y
CONFIG_NSS_DRV_MIRROR_ENABLE=y
CONFIG_NSS_DRV_MATCH_ENABLE=y
CONFIG_NSS_DRV_TUN6RD_ENABLE=y
CONFIG_NSS_DRV_TUNIPIP6_ENABLE=y
CONFIG_NSS_DRV_VXLAN_ENABLE=y
CONFIG_NSS_DRV_PVXLAN_ENABLE=y
CONFIG_NSS_DRV_WIFI_MESH_ENABLE=y
#ipq60xx 上 ath11k 没有 NSS 无线卸载补丁（那是 ipq807x 的），
#WIFIOFFLOAD 打开只会多编一段用不到的数据面，所以不开。
CONFIG_NSS_DRV_WIFIOFFLOAD_ENABLE=n

#NSS client 模块
CONFIG_PACKAGE_kmod-qca-nss-drv-bridge-mgr=y
CONFIG_PACKAGE_kmod-qca-nss-drv-vlan-mgr=y
CONFIG_PACKAGE_kmod-qca-nss-drv-eogremgr=y
CONFIG_PACKAGE_kmod-qca-nss-drv-gre=y
CONFIG_PACKAGE_kmod-qca-nss-drv-igs=y
CONFIG_PACKAGE_kmod-qca-nss-drv-qdisc=y
CONFIG_PACKAGE_kmod-qca-nss-drv-l2tpv2=y
CONFIG_PACKAGE_kmod-qca-nss-drv-lag-mgr=y
CONFIG_PACKAGE_kmod-qca-nss-drv-map-t=y
CONFIG_PACKAGE_kmod-qca-nss-drv-mirror=y
CONFIG_PACKAGE_kmod-qca-nss-drv-pppoe=y
CONFIG_PACKAGE_kmod-qca-nss-drv-pptp=y
CONFIG_PACKAGE_kmod-qca-nss-drv-tun6rd=y
CONFIG_PACKAGE_kmod-qca-nss-drv-tunipip6=y
CONFIG_PACKAGE_kmod-qca-nss-drv-vxlanmgr=y
CONFIG_PACKAGE_kmod-qca-nss-drv-pvxlanmgr=y
CONFIG_PACKAGE_kmod-qca-nss-drv-wifi-meshmgr=y
